Usage: tessera export [OPTIONS] --out <OUT> --format <FORMAT> [INPUT]

Arguments:
  [INPUT]

Options:
      --app-dir <APP_DIR>

      --query <QUERY>

      --json

      --out <OUT>

      --format <FORMAT>
          Output format. JPEG XL (jxl) is always lossless and sRGB-only [possible values: jpeg, png, tiff, avif, jxl]
      --bit-depth <BIT_DEPTH>
          AVIF 8/10/12 or TIFF/JPEG XL 8/16 bits per channel [default: 8]
      --avif-speed <AVIF_SPEED>
          AVIF encoding speed, 1 (slow) through 10 (fast) [default: 6]
      --quality <QUALITY>
          [default: 90]
      --max-file-bytes <MAX_FILE_BYTES>
          Maximum JPEG bytes including ICC and XMP (fails if unattainable)
      --watermark <WATERMARK>
          Watermark JSON file (text/font or PNG graphic, anchor, inset, opacity)
      --long-edge <LONG_EDGE>

      --fit <FIT>

      --color-space <COLOR_SPACE>
          [default: srgb] [possible values: srgb, p3, rec2020, prophoto]
      --sharpen <SHARPEN>
          [possible values: screen, matte, glossy]
      --metadata <METADATA>
          [default: all] [possible values: all, copyright, none]
      --name <NAME>
          [default: {name}-{seq}]
      --jobs <JOBS>

      --upscale <UPSCALE>
          Apply x2/x4 super-resolution before resize/sharpen (serial; ignores --jobs) [possible values: 2, 4]
  -h, --help
          Print help
